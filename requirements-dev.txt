# Development and test dependencies for Tyche.
#
# Install a CPU-only PyTorch first if you do not have a CUDA card, or pip will
# resolve the CUDA build from PyPI and download several extra gigabytes:
#
#   pip install torch --index-url https://download.pytorch.org/whl/cpu
#   pip install -r requirements-dev.txt
-r requirements.txt

# The test suite. Absent from requirements.txt: running Tyche never needs it.
pytest>=8.0

# Static analysis. It parses the sources rather than importing them, so it
# needs none of the runtime dependencies and runs in a second.
ruff>=0.5

# Parses .github/workflows/*.yml, so a broken release workflow is caught by
# the test suite rather than by the next tag silently publishing nothing.
# Imported hard by tests/test_release_workflow.py: a skip there would stop
# testing the file that publishes the releases, and say nothing about it.
pyyaml>=6.0

# Screenshot capture for docs/generate_screenshots.py. A documentation
# dependency, not a runtime one — Tyche never imports it.
pillow>=10.0
